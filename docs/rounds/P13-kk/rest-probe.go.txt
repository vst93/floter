package settingsui

import (
	"fmt"
	"image/png"
	"os"
	"testing"
)

// Temporary evidence probe; archived as text after before/after capture.
func TestKKRestingGeometry(t *testing.T) {
	for _, dark := range []bool{false, true} {
		for _, mode := range []string{"program", "script"} {
			a := declarationVisibilityApp(t, dark, 5)
			a.toolAdvanced = true
			if mode == "script" {
				a.toolMode, a.toolLanguage, a.toolScript = routeScript, "python", "print('sunny')"
			}
			tt := toolVisibilityFrame(t, a, dark)
			for i, y := range []float32{0, 600, a.Body.MaxY} {
				a.Body.Y = y
				tt.Frame()
				name := fmt.Sprintf("/tmp/floter-P13-kk-rest-%s-%s-%d-%s", mode, appearanceName(dark), i, os.Getenv("FLOTER_P13KK_REST"))
				f, err := os.Create(name + ".png")
				if err != nil {
					t.Fatal(err)
				}
				if err := png.Encode(f, tt.Image()); err != nil {
					t.Fatal(err)
				}
				if err := f.Close(); err != nil {
					t.Fatal(err)
				}
				geometry := fmt.Sprintf("scroll=%v max=%v\n", a.Body.Y, a.Body.MaxY)
				for _, label := range tt.Texts() {
					r, _ := tt.Find(label)
					geometry += fmt.Sprintf("%q %+v\n", label, r)
				}
				if err := os.WriteFile(name+".txt", []byte(geometry), 0600); err != nil {
					t.Fatal(err)
				}
			}
		}
	}
}
